\footnotesize
\begin{tcolorbox}[enhanced, colframe=yellow!100,

\setbeamertemplate{enumerate items}[default]
\begin{enumerate}
\only<1>{
\item \textbf{Honesty}:
Maintaining self-credibility, upholding the truth, not lying, not deceiving, and not cheating.
\item \textbf{Justice:}
Upholding the values of justice, being responsive, courteous, and non-discriminatory in treating every citizen.
\item \textbf{Trustworthiness:}
Being trustworthy and reliable in carrying out duties, and not misusing information, position, or facilities.
\item \textbf{Dignity:}
Upholding norms of decency and politeness, and treating everyone with respect and humanity.
\item \textbf{Responsibility:}
Being disciplined in completing responsibilities effectively and efficiently, and acting proactively for quality results.
}
\only<2>{
\setcounter{enumi}{5}
\item \textbf{Unity:}
Firmly holding the identity as the Indonesian nation, upholding tolerance for diversity, and working together to achieve common goals.
\item \textbf{Openness:}
Being willing to listen to others' opinions, accept academic information and findings from others, and be willing to share one's own knowledge.
\item \textbf{Academic Freedom:}
The obligation to maintain and advance science, and the freedom to express thoughts and opinions in academic forums.
\item \textbf{Compliance with Rules:}
Carrying out all activities within the UI environment by complying with all applicable laws and regulations.
}
\end{enumerate}

\end{tcolorbox}

\begin{frame}[fragile]{Course Learning Sub-Outcomes (Sub-CPMK) --- Before the Midterm}
\scriptsize
\begin{tcolorbox}[enhanced,width=5.9in,fonttitle=\scriptsize\bfseries,title=\scriptsize Sub-CPMK for Weeks 1--7]
\setbeamertemplate{enumerate items}[default]
    \setlength{\leftmargini}{2.15cm}
    \begin{itemize}
        \item[Sub-CPMK 1.1] Able to apply \textit{probability axioms}, \textit{conditional probability}, and \textit{Bayes' theorem} to compute the probability of an event (C3).
        \item[Sub-CPMK 1.2] Able to apply \textit{counting rules} to \textit{sequential experiments} and the \textit{probability mass function} to compute the probability of compound events and to characterise \textit{discrete random variables} (C3).
        \item[Sub-CPMK 1.3] Able to apply the \textit{probability density function} together with measures of centre and spread to characterise \textit{continuous random variables} (C3).
        \item[Sub-CPMK 1.4] Able to apply \textit{joint}, \textit{marginal}, and \textit{conditional distributions} to analyse the relationship between two \textit{random variables} (C3).
        \item[Sub-CPMK 1.5] Able to apply \textit{covariance} and the \textit{correlation coefficient} to measure the strength of the relationship between two \textit{random variables} and to distinguish it from \textit{independence} (C3).
        \item[Sub-CPMK 1.6] Able to apply \textit{transformation of random variables} and \textit{conditional expectation} to derive the distribution and moments of a new \textit{random variable} (C3).
    \end{itemize}
\end{tcolorbox}
\end{frame}

\begin{frame}[fragile]{Course Learning Sub-Outcomes (Sub-CPMK) --- After the Midterm}
\scriptsize
\begin{tcolorbox}[enhanced,width=5.9in,fonttitle=\scriptsize\bfseries,title=\scriptsize Sub-CPMK for Weeks 9--15]
\setbeamertemplate{enumerate items}[default]
    \setlength{\leftmargini}{2.15cm}
    \begin{itemize}
        \item[Sub-CPMK 1.7] Able to apply \textit{random vectors} and the \textit{covariance matrix} to predict one \textit{random variable} from other observed \textit{random variables} (C3).
        \item[Sub-CPMK 1.8] Able to apply the properties of \textit{sums} and \textit{sequences of random variables} to explain the behaviour of the \textit{sample mean} as the sample size grows (C3).
        \item[Sub-CPMK 1.9] Able to apply the concept of \textit{stochastic processes} to classify and characterise systems that evolve randomly over time (C3).
        \item[Sub-CPMK 1.10] Able to apply the \textit{Markov property} and the \textit{transition matrix} to compute the probability of the system state at step $n$ (C3).
        \item[Sub-CPMK 1.11] Able to apply \textit{state classification} and the \textit{stationary distribution} to determine the \textit{long-run behaviour} of \textit{Markov chains} (C3).
        \item[Sub-CPMK 1.12] Able to apply \textit{first step analysis} to compute the \textit{hitting time} and \textit{absorption probability} of \textit{Markov chains} (C3).
    \end{itemize}
\end{tcolorbox}
\end{frame}
