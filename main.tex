\setbeamertemplate{enumerate items}[default]
\begin{enumerate}
\only<1>{
\item \textbf{Honesty}:
Maintaining self-credibility, upholding the truth, not lying, not deceiving, and not cheating. 
\item \textbf{Justice:}
Upholding the values of justice, being responsive, courteous, and non-discriminatory in treating every citizen. 
\item \textbf{Trustworthiness:}
Being trustworthy and reliable in carrying out duties, and not misusing information, position, or facilities. 
\item \textbf{Dignity:}
Upholding norms of decency and politeness, and treating everyone with respect and humanity. 
\item \textbf{Responsibility:}
Being disciplined in completing responsibilities effectively and efficiently, and acting proactively for quality results. 
}
\only<2>{
\setcounter{enumi}{5}
\item \textbf{Unity:}
Firmly holding the identity as the Indonesian nation, upholding tolerance for diversity, and working together to achieve common goals. 
\item \textbf{Openness:}
Being willing to listen to others' opinions, accept academic information and findings from others, and be willing to share one's own knowledge. 
\item \textbf{Academic Freedom:}
The obligation to maintain and advance science, and the freedom to express thoughts and opinions in academic forums. 
\item \textbf{Compliance with Rules:}
Carrying out all activities within the UI environment by complying with all applicable laws and regulations.
}
\end{enumerate}

\begin{frame}[t]
\frametitle{Evaluation Tecniques}
    \begin{itemize}
        \item \textbf{Exercise:} It is a computer-based clozed questions (EMAS3) evaluation. It doesn't have specific weight but can be used to increase post-test and quiz score! (\textbf{Almost Every Week})
        \item \textbf{Post-Test 1-4:}  It is a computer-based clozed questions (EMAS3) evaluation, with (probably) several essay questions to answer (still in EMAS3, not in hardcopy book!). They have  10\% weight for each. (\textbf{Week 2\textsuperscript{nd}, 6\textsuperscript{th}, 10\textsuperscript{th}, 14\textsuperscript{th}})
        \item \textbf{Quiz 1-2:}  It is an essay questions in EMAS3 or in hardcopy book. They have 10\% weight for each. (\textbf{Week 4\textsuperscript{th}, 12\textsuperscript{th}})
        \item \textbf{Midterm Exam:}  It is an essay questions in hardcopy book. It has 20\% weight. (\textbf{8\textsuperscript{th} Week})
        \item \textbf{Final Term Exam:}  It is an essay + in hardcopy book, then oral questions. It has 20\% weight. (\textbf{15\textsuperscript{th}-16\textsuperscript{th} Week})
    \end{itemize}
\end{frame}
